# E2EE implementation plan

> **For the implementer:** do the tasks in order and tick off steps with checkboxes (`- [ ]` → `- [x]`). Each task is written TDD-style: failing test → minimal implementation → green test → commit. Commits are made only with the user's consent; if there is no consent to commit, still run the checks after each task.

**Goal:** the server stores and relays only ciphertext. The room title and messages are encrypted in the browser. The shared key comes from an ephemeral ECDH authenticated by the link secret and an optional password. A wrong password does not consume the invite.

**Architecture:** the protocol and client hardening are described in the [design document](../specs/2026-11-06-e2ee-design.md). On the Go side, `internal/chat` gets `envelope.go`, which checks the format of opaque fields. `Create` and `Join` take input structs with key shares and a verifier; `Send` takes ciphertext. In `internal/httpapi` only decoding and a new error code change. On the frontend, a separate ES module `web/crypto.js` (WebCrypto, no dependencies) is tested with `node --test`; non-extractable keys are stored in IndexedDB (`web/keystore.js`); `web/app.js` becomes a module. A CSP with Trusted Types, isolation headers and a frontend hash manifest (`cmd/webhash`) protect against XSS and make it possible to verify the deployed code.

**Stack:** Go 1.24+, `coder/websocket`, WebCrypto (P-256 ECDH, HKDF-SHA256, PBKDF2-SHA256, HMAC-SHA256, AES-256-GCM), Node 24 for tests only.

**Invariants that must not break:** snapshot and subscription registration under one mutex; no network I/O under the mutex; atomic invite consumption; deduplication by `(participant_id, client_message_id)`; TTL is never extended; limits and error codes from SPEC §4.

```go
package chat

import "encoding/base64"

// Sizes of opaque end-to-end encrypted fields. The server checks shape only;
// it never sees keys or plaintext. They mirror web/crypto.js.
const (
	NonceBytes     = 12
	TagBytes       = 16
	PadBlock       = 256
	MaxPadded      = 4096
	TitleBlobBytes = NonceBytes + PadBlock + TagBytes
	PublicKeyBytes = 65
	MACBytes       = 32
)

var b64 = base64.RawURLEncoding.Strict()

type KeyShare struct {
	PublicKey string `json:"public_key"`
	MAC       string `json:"mac"`
}

func decodeBlob(s string, max int) ([]byte, bool) {
	if len(s) > b64.EncodedLen(max) {
		return nil, false
	}
	b, err := b64.DecodeString(s)
	return b, err == nil
}

func validMessageBlob(s string) bool {
	b, ok := decodeBlob(s, NonceBytes+MaxPadded+TagBytes)
	n := len(b) - NonceBytes - TagBytes
	return ok && n >= PadBlock && n <= MaxPadded && n%PadBlock == 0
}

func validTitleBlob(s string) bool {
	b, ok := decodeBlob(s, TitleBlobBytes)
	return ok && len(b) == TitleBlobBytes
}

func validKeyShare(k KeyShare) bool {
	pub, ok := decodeBlob(k.PublicKey, PublicKeyBytes)
	return ok && len(pub) == PublicKeyBytes && pub[0] == 0x04 && valid32(k.MAC)
}

func valid32(s string) bool {
	b, ok := decodeBlob(s, MACBytes)
	return ok && len(b) == MACBytes
}
```

func (s *Service) Join(ctx context.Context, in JoinInput) (Credentials, error) {
	if ctx == nil {
		ctx = context.Background()
	}
	if err := ctx.Err(); err != nil {
		return Credentials{}, err
	}
	if !valid32(in.JoinProof) || !validKeyShare(in.KeyShare) {
		return Credentials{}, ErrInput
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.expireLocked()
	r := s.invites[digest(in.InviteToken)]
	if r == nil || s.closed {
		return Credentials{}, ErrUnavailable
	}
	proof := digest(in.JoinProof)
	if subtle.ConstantTimeCompare(proof[:], r.verifier[:]) != 1 {
		// A wrong proof must not consume the seat, but online guessing is bounded.
		r.failedJoins++
		if r.failedJoins >= MaxJoinAttempts {
			delete(s.invites, r.invite)
			s.broadcast(r, Event{Type: "invite_revoked", Payload: struct{}{}})
		}
		return Credentials{}, ErrJoinProof
	}
	delete(s.invites, r.invite)
	c := s.addSession(r, "guest", in.KeyShare)
	s.broadcast(r, Event{Type: "participants", Payload: append([]Participant(nil), r.participants...)})
	return c, nil
}

// End-to-end encryption for nullchat. The server only ever sees the outputs of
// this module; see docs/superpowers/specs/2026-11-06-e2ee-design.md.
const subtle = globalThis.crypto.subtle;
const enc = new TextEncoder();
const dec = new TextDecoder('utf-8', { fatal: true });

export const VERSION = 'nullchat-e2ee-v1';
export const PBKDF2_ITERATIONS = 600_000;
export const PAD_BLOCK = 256;
export const MAX_TEXT_BYTES = 4000;
const NONCE_BYTES = 12;
const EC = { name: 'ECDH', namedCurve: 'P-256' };

const label = (purpose) => enc.encode(`${VERSION} ${purpose}`);
const concat = (...parts) => {
  const out = new Uint8Array(parts.reduce((n, p) => n + p.length, 0));
  let offset = 0;
  for (const p of parts) { out.set(p, offset); offset += p.length; }
  return out;
};

export function toB64(bytes) {
  let s = '';
  for (const b of new Uint8Array(bytes)) s += String.fromCharCode(b);
  return btoa(s).replaceAll('+', '-').replaceAll('/', '_').replace(/=+$/, '');
}

export function fromB64(text) {
  if (typeof text !== 'string' || !/^[A-Za-z0-9_-]*$/.test(text)) throw new TypeError('invalid base64url');
  const s = atob(text.replaceAll('-', '+').replaceAll('_', '/') + '='.repeat((4 - text.length % 4) % 4));
  return Uint8Array.from(s, (c) => c.charCodeAt(0));
}

export const randomSecret = () => toB64(crypto.getRandomValues(new Uint8Array(32)));

// ISO/IEC 7816-4: 0x80 then zeros up to the next multiple of the block size.
export function pad(bytes, block = PAD_BLOCK) {
  const out = new Uint8Array(Math.ceil((bytes.length + 1) / block) * block);
  out.set(bytes);
  out[bytes.length] = 0x80;
  return out;
}

export function unpad(bytes) {
  let i = bytes.length - 1;
  while (i >= 0 && bytes[i] === 0) i--;
  if (i < 0 || bytes[i] !== 0x80) throw new Error('invalid padding');
  return bytes.subarray(0, i);
}

async function hkdf(ikm, info, salt = label('salt')) {
  const key = await subtle.importKey('raw', ikm, 'HKDF', false, ['deriveBits']);
  return new Uint8Array(await subtle.deriveBits({ name: 'HKDF', hash: 'SHA-256', salt, info }, key, 256));
}

const aesKey = (bits) => subtle.importKey('raw', bits, 'AES-GCM', false, ['encrypt', 'decrypt']);
const hmacKey = (bits, usages) => subtle.importKey('raw', bits, { name: 'HMAC', hash: 'SHA-256' }, false, usages);

// Returns '' without a password so the stored session carries no password material.
export async function passwordKey(secret, password) {
  if (!password) return '';
  const key = await subtle.importKey('raw', enc.encode(password.normalize('NFC')), 'PBKDF2', false, ['deriveBits']);
  const salt = concat(label('password'), fromB64(secret));
  return toB64(await subtle.deriveBits({ name: 'PBKDF2', hash: 'SHA-256', salt, iterations: PBKDF2_ITERATIONS }, key, 256));
}

export async function roomSecrets(secret, pwKey = '') {
  const s = fromB64(secret);
  if (s.length !== 32) throw new Error('invalid room secret');
  const pw = pwKey ? fromB64(pwKey) : new Uint8Array();
  const ikm = concat(s, Uint8Array.of(pw.length ? 1 : 0), pw);
  const [title, mac, join, roomSalt] = await Promise.all(['title', 'handshake', 'join', 'room-salt'].map((p) => hkdf(ikm, label(p))));
  const joinKey = await hmacKey(join, ['sign']);
  return {
    titleKey: await aesKey(title),
    macKey: await hmacKey(mac, ['sign', 'verify']),
    roomSalt,
    verifier: toB64(await subtle.sign('HMAC', joinKey, label('join'))),
  };
}

// The private key is non-extractable: script injected into the page can use it while
// the tab is open but cannot copy it. ECDH public keys are always exportable.
export async function generateKeyPair() {
  const pair = await subtle.generateKey(EC, false, ['deriveBits']);
  return { privateKey: pair.privateKey, publicKey: toB64(await subtle.exportKey('raw', pair.publicKey)) };
}

const shareData = (role, publicKey) => concat(label(`share ${role}`), fromB64(publicKey));

export async function signShare(secrets, role, publicKey) {
  return { public_key: publicKey, mac: toB64(await subtle.sign('HMAC', secrets.macKey, shareData(role, publicKey))) };
}

export async function verifyShare(secrets, role, share) {
  try {
    if (!(await subtle.verify('HMAC', secrets.macKey, fromB64(share.mac), shareData(role, share.public_key)))) return false;
    await subtle.importKey('raw', fromB64(share.public_key), EC, false, []); // rejects points not on the curve
    return true;
  } catch { return false; }
}

export async function messageKey(secrets, privateKey, creatorPublic, guestPublic, selfRole) {
  const peer = await subtle.importKey('raw', fromB64(selfRole === 'creator' ? guestPublic : creatorPublic), EC, false, []);
  const shared = new Uint8Array(await subtle.deriveBits({ name: 'ECDH', public: peer }, privateKey, 256));
  const info = concat(label('messages'), fromB64(creatorPublic), fromB64(guestPublic));
  return aesKey(await hkdf(shared, info, secrets.roomSalt));
}

export async function safetyCode(creatorPublic, guestPublic) {
  const view = new DataView(await subtle.digest('SHA-256', concat(label('safety'), fromB64(creatorPublic), fromB64(guestPublic))));
  return [0, 4, 8].map((o) => String(view.getUint32(o) % 10000).padStart(4, '0')).join(' ');
}

async function seal(key, plaintext, aad) {
  const iv = crypto.getRandomValues(new Uint8Array(NONCE_BYTES));
  return toB64(concat(iv, new Uint8Array(await subtle.encrypt({ name: 'AES-GCM', iv, additionalData: aad }, key, plaintext))));
}

async function open(key, blob, aad) {
  const b = fromB64(blob);
  return new Uint8Array(await subtle.decrypt({ name: 'AES-GCM', iv: b.subarray(0, NONCE_BYTES), additionalData: aad }, key, b.subarray(NONCE_BYTES)));
}

export async function encryptTitle(secrets, title) {
  const bytes = enc.encode(title);
  if (bytes.length >= PAD_BLOCK) throw new RangeError('title too long');
  return seal(secrets.titleKey, pad(bytes), label('title'));
}

export async function decryptTitle(secrets, blob) {
  return dec.decode(unpad(await open(secrets.titleKey, blob, label('title'))));
}

const messageAad = ({ roomId, senderId, clientId }) => enc.encode(JSON.stringify([`${VERSION} message`, roomId, senderId, clientId]));

export async function encryptMessage(key, { roomId, senderId, clientId, counter, text }) {
  const body = enc.encode(text);
  if (body.length > MAX_TEXT_BYTES) throw new RangeError('text too long');
  const header = new Uint8Array(5);
  header[0] = 1;
  new DataView(header.buffer).setUint32(1, counter);
  return seal(key, pad(concat(header, body)), messageAad({ roomId, senderId, clientId }));
}

export async function decryptMessage(key, { roomId, senderId, clientId, ciphertext }) {
  const plain = unpad(await open(key, ciphertext, messageAad({ roomId, senderId, clientId })));
  if (plain.length < 5 || plain[0] !== 1) throw new Error('unsupported message version');
  return { counter: new DataView(plain.buffer, plain.byteOffset).getUint32(1), text: dec.decode(plain.subarray(5)) };
}

// Messages must be in sequence order. A counter that does not increase is a replay
// or reordering; a jump means messages were never sent or were dropped by the server.
export function checkCounters(messages) {
  const last = new Map();
  return messages.map(({ senderId, counter }) => {
    const prev = last.get(senderId) ?? 0;
    if (counter <= prev) return { gap: 0, replay: true };
    last.set(senderId, counter);
    return { gap: counter - prev - 1, replay: false };
  });
}

// Stores non-extractable CryptoKeys between reloads. IndexedDB is shared by the
// origin and outlives the tab, so records carry expires_at and are swept.
const DB_NAME = 'nullchat';
const STORE = 'keys';

function openDb() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, 1);
    request.onupgradeneeded = () => request.result.createObjectStore(STORE);
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}

async function run(mode, action) {
  const db = await openDb();
  try {
    return await new Promise((resolve, reject) => {
      const tx = db.transaction(STORE, mode);
      const request = action(tx.objectStore(STORE));
      tx.oncomplete = () => resolve(request?.result);
      tx.onerror = tx.onabort = () => reject(tx.error);
    });
  } finally { db.close(); }
}

export const keyId = (roomId, participantId) => `${roomId}:${participantId}`;
export const putKeys = (id, record) => run('readwrite', (store) => store.put(record, id));
export const getKeys = (id) => run('readonly', (store) => store.get(id));
export const deleteKeys = (id) => run('readwrite', (store) => store.delete(id));

export function sweepExpired(now = Date.now()) {
  return run('readwrite', (store) => {
    const cursor = store.openCursor();
    cursor.onsuccess = () => {
      const c = cursor.result;
      if (!c) return;
      if (!(Date.parse(c.value?.expires_at) > now)) c.delete();
      c.continue();
    };
    return null;
  });
}

  if (!state.messageKey) return;
  for (const m of state.room.messages) {
    if (state.plain.has(m.id)) continue;
    try { state.plain.set(m.id, await e2ee.decryptMessage(state.messageKey, { roomId: state.room.id, senderId: m.sender_id, clientId: m.client_message_id, ciphertext: m.ciphertext })); }
    catch { state.plain.set(m.id, { error: true }); }
  }
  const ok = state.room.messages.filter((m) => !state.plain.get(m.id)?.error);
  const flags = e2ee.checkCounters(ok.map((m) => ({ senderId: m.sender_id, counter: state.plain.get(m.id).counter })));
  state.flags = new Map(ok.map((m, i) => [m.id, flags[i]]));
  const own = ok.filter((m) => m.sender_id === state.self.id).map((m) => state.plain.get(m.id).counter);
  state.e2ee.next_counter = Math.max(state.e2ee.next_counter, ...own.map((c) => c + 1));
}

    state.pending.delete(message.client_message_id);
    state.failed.delete(message.client_message_id);

}

```go
// Manifest returns the SHA-256 of every embedded asset and a digest of the sorted
// "hash  name" lines, so a deployment can be compared with a published release.
func Manifest() (map[string]string, string, error) {
	files := map[string]string{}
	err := fs.WalkDir(assets, ".", func(path string, d fs.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		data, err := assets.ReadFile(path)
		if err != nil {
			return err
		}
		sum := sha256.Sum256(data)
		files[path] = hex.EncodeToString(sum[:])
		return nil
	})
	if err != nil {
		return nil, "", err
	}
	names := make([]string, 0, len(files))
	for name := range files {
		names = append(names, name)
	}
	sort.Strings(names)
	var lines strings.Builder
	for _, name := range names {
		fmt.Fprintf(&lines, "%s  %s\n", files[name], name)
	}
	total := sha256.Sum256([]byte(lines.String()))
	return files, hex.EncodeToString(total[:]), nil
}
```
